Node: MaterialExpressionCustom_0 | Desc: 'CalcRipple'
Includes: []
Inputs: ['RippleTex', 'WorldPos', 'ServerTime']
Code:
// 이번 정점의 최종 Z축 이동 Offset 누적 변수
float TotalHeight = 0.0;

// 최대 32개의 파동에 대해 반복
for (int i = 0; i < 32; ++i)
{
    // C++에서 동적으로 생성된 RippleTex로부터 파동 정보 파싱
    // Row 0: Origin.X, Origin.Y, StartTime, InitialAmplitude (RGBA 순서)
    float4 Row1 = RippleTex.Load(int3(i, 0, 0));
    
    // Row 1: WaveSpeed, DecayRate, WaveLength, ExpireTime (RGBA 순서)
    float4 Row2 = RippleTex.Load(int3(i, 1, 0));
    
    float2 Origin = Row1.xy;
    float StartTime = Row1.z;
    float InitialAmplitude = Row1.w;
    
    float WaveSpeed = Row2.x;
    float DecayRate = Row2.y;
    float WaveLength = Row2.z;
    float ExpireTime = Row2.w;
    
    // 비활성화된 슬롯이거나 이미 소멸 시간이 지난 파동은 스킵
    if (InitialAmplitude <= 0.0 || ServerTime >= ExpireTime)
    {
        continue;
    }
    
    // 예약된 파동이므로 스킵 (네트워크 지연으로 예약 파동이 생길 수 있음)
    float dt = ServerTime - StartTime;
    if (dt < 0.0)
    {
        continue;
    }
    
    float d = distance(WorldPos, Origin); // 정점과 파원사이의 거리
    float R = WaveSpeed * dt; // 파동의 현재까지의 이동거리
    float W = WaveLength * 2.0; // 파동의 반치 폭
    
    // Bounding 영역 컬링: 선두 전파 범위 밖의 정점 연산 제외
    if (d > R + W || (R > W && d < R - W))
    {
        // d > R + W : 정점이 너무 멀리 있어서 파동이 닿지 않은 경우
        // R > W && d < R - W
        continue;
    }
    
    // 시간 경과에 따른 진폭 감쇄
    float Decay = exp(-DecayRate * dt);
    
    // 파동 선두와 정점 사이의 거리
    float DistFromWavefront = abs(d - R);

    // 거리: 0 ~ W | Envelope 1 ~ 0
    // 파동 선두의 영향 범위 내에서의 부드러운 감쇄
    float Envelope = 1.0 - smoothstep(0.0, W, DistFromWavefront);
    
    // 위상(Phase) : 파동 한 주기 내부의 어디에 있는가? (라디안)
    float Phase = (d - R) / WaveLength * 2.0 * 3.141592;

    // 최종 높이 계산
    float Height = InitialAmplitude * Decay * cos(Phase) * Envelope;
    
    TotalHeight += Height;
}

return TotalHeight;

Node: MaterialExpressionCustom_16 | Desc: 'CalcOceanFoam'
Includes: []
Inputs: ['ScaleFoam', 'WhiteFoam', 'WaveHeight', 'WaveSteepness', 'FoamColor', 'TranslucentColor', 'SlopeBounds', 'CrestBounds', 'SSSIntensity']
Code:
// 1. 넓은 주변부 마스크 
float slopeHeight = smoothstep(SlopeBounds.r, SlopeBounds.g, WaveHeight);
float slopeSteepness = smoothstep(SlopeBounds.b, SlopeBounds.a, WaveSteepness);
float slopeMask = saturate(slopeHeight * slopeSteepness * slopeSteepness);

// 2. 뾰족한 정상부 마스크 
float crestHeight = smoothstep(CrestBounds.r, CrestBounds.g, WaveHeight);
float crestSteepness = smoothstep(CrestBounds.b, CrestBounds.a, WaveSteepness);
float crestMask = saturate(crestHeight * crestSteepness);

// 3. 텍스쳐에 직접 색상 입히기 (핵심 변경점!)
// 비늘(그물) 텍스쳐의 밝은 선 부분만 에메랄드 빛으로 빛나게 만듭니다.
float3 emeraldNetFoam = ScaleFoam * TranslucentColor * SSSIntensity;

// 하얀 거품 텍스쳐는 퓨어한 흰색으로 빛나게 둡니다.
float3 whiteCrestFoam = WhiteFoam * FoamColor;

// 4. 레이어링
// 넓은 주변부에는 에메랄드 그물 거품을 깔아줍니다.
float3 baseLayer = emeraldNetFoam * slopeMask;

// 정상부(crestMask)로 갈수록 새하얀 거품으로 완벽하게 덮어버립니다.
float3 finalFoam = lerp(baseLayer, whiteCrestFoam, crestMask);

return finalFoam;
